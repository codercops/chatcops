title: Cloudflare Workers Adapter
description: Deploy ChatCops as a Cloudflare Worker.
---

The Cloudflare adapter creates a handler compatible with the Workers runtime.

## Setup

Create a Worker at `src/index.ts`:

```typescript
import { chatcopsCloudflareHandler } from '@chatcops/server';

// Built on the first request rather than at import, because the API key is a
// Worker secret that only exists inside a request. Caching the handler here is
// what keeps conversation history and rate limiting alive across requests.
let handler: ReturnType<typeof chatcopsCloudflareHandler> | undefined;
function getHandler(env: Record<string, string>) {
  handler ??= chatcopsCloudflareHandler({
    provider: {
      type: 'claude',
      apiKey: env.ANTHROPIC_API_KEY,
    },
    systemPrompt: 'You are a helpful assistant.',
    allowedOrigins: ['*'],
    // Signs the session tokens the widget gets from /session (a Worker sees every path).
    sessionSecrets: [env.CHATCOPS_SESSION_SECRET],
  });
  return handler;
}

export default {
  async fetch(
    request: Request,
    env: Record<string, string>
  ): Promise<Response> {
    return getHandler(env)(request);
  },
};
```

:::note[History and rate limits]
Conversation history and rate limiting live in memory on the isolate, not in a
shared store. They are not shared between isolates and they do not survive a
restart, so a deploy or a scale event resets both.
:::

## Environment Variables

Set secrets using Wrangler:

```bash
npx wrangler secret put ANTHROPIC_API_KEY
npx wrangler secret put CHATCOPS_SESSION_SECRET   # paste the output of: openssl rand -base64 32
```

Or in `wrangler.toml` for non-secret values:

```toml
[vars]
SYSTEM_PROMPT = "You are a helpful assistant."
```

## Widget Configuration

```html
<script
  src="https://cdn.jsdelivr.net/npm/@chatcops/widget/dist/chatcops.min.js"
  data-api-url="https://chatcops.your-domain.workers.dev"
></script>
```

## Full Example

```typescript
import { chatcopsCloudflareHandler } from '@chatcops/server';
import { LeadCaptureTool } from '@chatcops/core';

let handler: ReturnType<typeof chatcopsCloudflareHandler> | undefined;
function getHandler(env: Record<string, string>) {
  handler ??= chatcopsCloudflareHandler({
    provider: {
      type: 'gemini',
      apiKey: env.GOOGLE_AI_API_KEY,
      model: 'gemini-3.5-flash-lite',
    },
    systemPrompt: env.SYSTEM_PROMPT || 'You are a helpful assistant.',
    sessionSecrets: [env.CHATCOPS_SESSION_SECRET],
    tools: [
      new LeadCaptureTool(async (lead) => {
        // Use Cloudflare KV, D1, or external webhook
        await fetch(env.WEBHOOK_URL, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(lead),
        });
      }),
    ],
    allowedOrigins: ['*'],
    rateLimit: { maxRequests: 20, windowMs: 60_000 },
  });
  return handler;
}

export default {
  async fetch(request: Request, env: Record<string, string>): Promise<Response> {
    return getHandler(env)(request);
  },
};
```

## Notes

- Cloudflare Workers have a 30-second CPU time limit (Bundled plan) or 15 minutes (Unbound)
- SSE streaming works natively on Workers
- Use Cloudflare KV or D1 for persistent storage (conversation history, analytics)
- The `env` object provides access to secrets and bindings
